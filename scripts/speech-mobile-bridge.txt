function safeSpeechError(status,data){
 var detail=typeof data?.message==='string'?data.message:'',code=typeof data?.code==='string'?data.code:'HTTP_'+status,stage=typeof data?.stage==='string'?data.stage:'';
 if(status===401&&code==='speech_access_required')detail='在「網站預設連線」輸入 ADMI_SITE_ACCESS_CODE；或在語音設定輸入原有 ADMI_SPEECH_ACCESS_TOKEN。不是 Google API key。';
 if(code==='google_credentials_missing_or_invalid')detail='Vercel 尚未讀到有效的 GOOGLE_SERVICE_ACCOUNT_JSON；請貼完整 JSON，選對 Production 後重新部署。';
 if(stage==='stt_recognize')detail+=' 請核對 STT 模型、地區、語言及錄音格式；翻譯支援不代表 STT 支援。';
 return (stage?'['+stage+'] ':'')+code+(detail?' · '+detail:'');
}
async function callSpeechV2(op,payload){
 var headers={'content-type':'application/json'},url;
 if(HOSTED)url='/ext/v1/google-speech/speech';
 else if(speechConfig){url=speechConfig.baseUrl+'/api/speech';headers.authorization='Bearer '+speechConfig.token;}
 else if(USE_SERVER_API){url='/api/speech';headers=websiteHeaders();}
 else throw new Error('請先設定 Speech Backend Base URL 及後端存取碼 / Configure speech backend access first.');
 var body=JSON.stringify(Object.assign({op:op},payload||{}));
 if(HOSTED&&new TextEncoder().encode(body).length>60000)throw new Error('錄音超過學校代理大小限制，未有上傳；請縮短錄音。');
 var controller=new AbortController(),timer=setTimeout(function(){controller.abort();},80000);
 try{var r=await fetch(url,{method:'POST',headers:headers,body:body,credentials:HOSTED||USE_SERVER_API?'same-origin':'omit',signal:controller.signal,referrerPolicy:'strict-origin-when-cross-origin'});var d=await r.json().catch(function(){return null;});if(!r.ok)throw new Error(safeSpeechError(r.status,d));if(!d)throw new Error('語音後端未回傳 JSON，請檢查網址／部署。');return d;}
 catch(e){if(e.name==='AbortError')throw new Error('語音請求逾時，請稍後自行再試。');if(e instanceof TypeError)throw new Error('語音後端連線／CORS 失敗，請檢查網址及允許來源。');throw e;}finally{clearTimeout(timer);}
}

var baselineHeightV2=window.innerHeight;

}
